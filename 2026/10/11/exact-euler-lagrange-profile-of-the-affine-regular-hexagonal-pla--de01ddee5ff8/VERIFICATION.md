\[
L_{\mathrm{YJ}}(\lambda,\mu,X_H)
=
\frac{2+r^2+r\sqrt{r^2+4}}{2(1+r^2)},
\qquad
r=\frac{\min\{\lambda,\mu\}}{\max\{\lambda,\mu\}},
\]
for the affine-regular hexagonal norm
\[
N(s,t)=\max\{|s|,|t|,|s-t|\}.
