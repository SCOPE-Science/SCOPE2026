\[
\deg\operatorname{Split}_d(\mathbb P^n)

\frac{(nd)!}{d!(n!)^d}.
\]

\[
\chi_{\mathrm{top}}\!\left(\operatorname{Split}_d(\mathbb P^n)\right)

\binom{n+d}{d}.
